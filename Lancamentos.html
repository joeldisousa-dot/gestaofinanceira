/************************************************************ * CONTROLE FINANCEIRO * GOOGLE APPS SCRIPT - Code.gs * * ABA LANCAMENTOS: * ID | Competência | Tipo | Descricao | Valor | Vencimento | Baixado * * ABA DESCRICOES: * Descricao ************************************************************/ /************************************************************ * CONFIGURAÇÃO ************************************************************/ const NOME_ABA_LANCAMENTOS = "Lancamentos"; const NOME_ABA_DESCRICOES = "Descricoes"; const COLUNAS_LANCAMENTOS = [ "ID", "Competência", "Tipo", "Descricao", "Valor", "Vencimento", "Baixado" ]; /************************************************************ * API PRINCIPAL ************************************************************/ function doGet(e) { try { const parametros = e && e.parameter ? e.parameter : {}; const acao = String( parametros.action || "" ).trim(); if (!acao) { return respostaJson({ sucesso: true, mensagem: "API do Controle Financeiro funcionando.", acoes: [ "listarLancamentos", "incluirLancamento", "editarLancamento", "baixarLancamento", "excluirLancamento", "obterResumo", "listarDescricoes" ] }); } switch (acao) { case "listarLancamentos": return respostaJson( listarLancamentos(parametros) ); case "incluirLancamento": return respostaJson( incluirLancamento(parametros) ); case "editarLancamento": return respostaJson( editarLancamento(parametros) ); case "baixarLancamento": return respostaJson( baixarLancamento(parametros) ); case "excluirLancamento": return respostaJson( excluirLancamento(parametros) ); case "obterResumo": return respostaJson( obterResumo(parametros) ); case "listarDescricoes": return respostaJson( listarDescricoes() ); default: throw new Error( "Ação não reconhecida: " + acao ); } } catch (erro) { console.error( erro ); return respostaJson({ sucesso: false, erro: erro.message || String(erro) }); } } /************************************************************ * RESPOSTA JSON ************************************************************/ function respostaJson(objeto) { return ContentService .createTextOutput( JSON.stringify(objeto) ) .setMimeType( ContentService.MimeType.JSON ); } /************************************************************ * PLANILHA ************************************************************/ function obterPlanilha() { const planilha = SpreadsheetApp.getActiveSpreadsheet(); if (!planilha) { throw new Error( "Não foi possível acessar a planilha." ); } return planilha; } /************************************************************ * ABA LANCAMENTOS ************************************************************/ function obterAbaLancamentos() { const planilha = obterPlanilha(); const aba = planilha.getSheetByName( NOME_ABA_LANCAMENTOS ); if (!aba) { throw new Error( 'A aba "' + NOME_ABA_LANCAMENTOS + '" não foi encontrada.' ); } validarEstruturaLancamentos( aba ); return aba; } /************************************************************ * VALIDAR ESTRUTURA ************************************************************/ function validarEstruturaLancamentos( aba ) { const ultimaColuna = aba.getLastColumn(); if (ultimaColuna < COLUNAS_LANCAMENTOS.length) { throw new Error( 'A aba "' + NOME_ABA_LANCAMENTOS + '" possui estrutura incompatível. ' + "Colunas esperadas: " + COLUNAS_LANCAMENTOS.join(", ") ); } const cabecalho = aba .getRange( 1, 1, 1, COLUNAS_LANCAMENTOS.length ) .getValues()[0]; const normalizadas = cabecalho.map( function(valor) { return normalizarCabecalho( valor ); } ); const ausentes = []; COLUNAS_LANCAMENTOS.forEach( function(coluna) { if ( normalizadas.indexOf( normalizarCabecalho(coluna) ) === -1 ) { ausentes.push( coluna ); } } ); if (ausentes.length > 0) { throw new Error( 'A aba "' + NOME_ABA_LANCAMENTOS + '" possui estrutura incompatível. ' + "Colunas ausentes: " + ausentes.join(", ") ); } } /************************************************************ * NORMALIZAR CABEÇALHO ************************************************************/ function normalizarCabecalho( valor ) { return String( valor || "" ) .trim() .toLowerCase() .normalize("NFD") .replace( /[\u0300-\u036f]/g, "" ); } /************************************************************ * MAPA DAS COLUNAS ************************************************************/ function obterMapaColunas( aba ) { const ultimaColuna = aba.getLastColumn(); const cabecalho = aba .getRange( 1, 1, 1, ultimaColuna ) .getValues()[0]; const mapa = {}; cabecalho.forEach( function(nome, indice) { mapa[ normalizarCabecalho(nome) ] = indice + 1; } ); return mapa; } /************************************************************ * LISTAR LANÇAMENTOS ************************************************************/ function listarLancamentos( parametros ) { const competencia = String( parametros.competencia || "" ).trim(); if (!competencia) { throw new Error( "Informe a competência." ); } const aba = obterAbaLancamentos(); const mapa = obterMapaColunas( aba ); const ultimaLinha = aba.getLastRow(); if (ultimaLinha < 2) { return { sucesso: true, lista: [] }; } const ultimaColuna = aba.getLastColumn(); const dados = aba .getRange( 2, 1, ultimaLinha - 1, ultimaColuna ) .getValues(); const lista = []; dados.forEach( function(linha) { const id = linha[ mapa["id"] - 1 ]; if ( id === "" || id === null || id === undefined ) { return; } const competenciaLinha = converterCompetencia( linha[ mapa["competencia"] - 1 ] ); if ( competenciaLinha !== competencia ) { return; } lista.push({ id: String(id), competencia: competenciaLinha, tipo: String( linha[ mapa["tipo"] - 1 ] || "" ), descricao: String( linha[ mapa["descricao"] - 1 ] || "" ), valor: converterNumero( linha[ mapa["valor"] - 1 ] ), vencimento: formatarData( linha[ mapa["vencimento"] - 1 ] ), baixado: normalizarBaixado( linha[ mapa["baixado"] - 1 ] ) }); } ); return { sucesso: true, lista: lista }; } /************************************************************ * INCLUIR LANÇAMENTO ************************************************************/ function incluirLancamento( parametros ) { const competencia = String( parametros.competencia || "" ).trim(); const tipo = String( parametros.tipo || "" ).trim(); const descricao = String( parametros.descricao || "" ).trim(); const valor = converterNumero( parametros.valor ); const vencimento = String( parametros.vencimento || "" ).trim(); validarLancamento( competencia, tipo, descricao, valor, vencimento ); const aba = obterAbaLancamentos(); const mapa = obterMapaColunas( aba ); const id = gerarNovoId( aba, mapa ); const linha = new Array( aba.getLastColumn() ).fill(""); linha[ mapa["id"] - 1 ] = id; linha[ mapa["competencia"] - 1 ] = competencia; linha[ mapa["tipo"] - 1 ] = tipo; linha[ mapa["descricao"] - 1 ] = descricao; linha[ mapa["valor"] - 1 ] = valor; linha[ mapa["vencimento"] - 1 ] = converterDataPlanilha( vencimento ); linha[ mapa["baixado"] - 1 ] = "Não"; aba.appendRow( linha ); return { sucesso: true, mensagem: "Lançamento incluído com sucesso.", id: String(id) }; } /************************************************************ * EDITAR LANÇAMENTO ************************************************************/ function editarLancamento( parametros ) { const id = String( parametros.id || "" ).trim(); if (!id) { throw new Error( "Informe o ID do lançamento." ); } const competencia = String( parametros.competencia || "" ).trim(); const tipo = String( parametros.tipo || "" ).trim(); const descricao = String( parametros.descricao || "" ).trim(); const valor = converterNumero( parametros.valor ); const vencimento = String( parametros.vencimento || "" ).trim(); validarLancamento( competencia, tipo, descricao, valor, vencimento ); const aba = obterAbaLancamentos(); const mapa = obterMapaColunas( aba ); const linhaNumero = localizarLinhaPorId( aba, mapa, id ); if (!linhaNumero) { throw new Error( "Lançamento #" + id + " não encontrado." ); } /* * Mantém o valor atual de "Baixado". */ const baixadoAtual = aba .getRange( linhaNumero, mapa["baixado"] ) .getValue(); aba .getRange( linhaNumero, mapa["competencia"] ) .setValue( competencia ); aba .getRange( linhaNumero, mapa["tipo"] ) .setValue( tipo ); aba .getRange( linhaNumero, mapa["descricao"] ) .setValue( descricao ); aba .getRange( linhaNumero, mapa["valor"] ) .setValue( valor ); aba .getRange( linhaNumero, mapa["vencimento"] ) .setValue( converterDataPlanilha( vencimento ) ); aba .getRange( linhaNumero, mapa["baixado"] ) .setValue( baixadoAtual ); return { sucesso: true, mensagem: "Lançamento #" + id + " alterado com sucesso." }; } /************************************************************ * BAIXAR LANÇAMENTO ************************************************************/ function baixarLancamento( parametros ) { const id = String( parametros.id || "" ).trim(); if (!id) { throw new Error( "Informe o ID do lançamento." ); } const aba = obterAbaLancamentos(); const mapa = obterMapaColunas( aba ); const linhaNumero = localizarLinhaPorId( aba, mapa, id ); if (!linhaNumero) { throw new Error( "Lançamento #" + id + " não encontrado." ); } aba .getRange( linhaNumero, mapa["baixado"] ) .setValue( "Sim" ); return { sucesso: true, mensagem: "Lançamento #" + id + " baixado com sucesso." }; } /************************************************************ * EXCLUIR LANÇAMENTO ************************************************************/ function excluirLancamento( parametros ) { const id = String( parametros.id || "" ).trim(); if (!id) { throw new Error( "Informe o ID do lançamento." ); } const aba = obterAbaLancamentos(); const mapa = obterMapaColunas( aba ); const linhaNumero = localizarLinhaPorId( aba, mapa, id ); if (!linhaNumero) { throw new Error( "Lançamento #" + id + " não encontrado." ); } aba.deleteRow( linhaNumero ); return { sucesso: true, mensagem: "Lançamento #" + id + " excluído com sucesso." }; } /************************************************************ * LOCALIZAR ID ************************************************************/ function localizarLinhaPorId( aba, mapa, id ) { const ultimaLinha = aba.getLastRow(); if (ultimaLinha < 2) { return null; } const valores = aba .getRange( 2, mapa["id"], ultimaLinha - 1, 1 ) .getValues(); for ( let i = 0; i < valores.length; i++ ) { if ( String( valores[i][0] ).trim() === String(id).trim() ) { return i + 2; } } return null; } /************************************************************ * GERAR NOVO ID ************************************************************/ function gerarNovoId( aba, mapa ) { const ultimaLinha = aba.getLastRow(); if (ultimaLinha < 2) { return 1; } const valores = aba .getRange( 2, mapa["id"], ultimaLinha - 1, 1 ) .getValues(); let maiorId = 0; valores.forEach( function(linha) { const numero = parseInt( String( linha[0] ).trim(), 10 ); if ( !isNaN(numero) && numero > maiorId ) { maiorId = numero; } } ); return maiorId + 1; } /************************************************************ * RESUMO ************************************************************/ function obterResumo( parametros ) { const competencia = String( parametros.competencia || "" ).trim(); if (!competencia) { throw new Error( "Informe a competência." ); } const aba = obterAbaLancamentos(); const mapa = obterMapaColunas( aba ); const ultimaLinha = aba.getLastRow(); let receber = 0; let pagar = 0; let recebido = 0; let pago = 0; if (ultimaLinha >= 2) { const dados = aba .getRange( 2, 1, ultimaLinha - 1, aba.getLastColumn() ) .getValues(); dados.forEach( function(linha) { const competenciaLinha = converterCompetencia( linha[ mapa["competencia"] - 1 ] ); if ( competenciaLinha !== competencia ) { return; } const tipo = String( linha[ mapa["tipo"] - 1 ] || "" ) .trim() .toLowerCase(); const valor = converterNumero( linha[ mapa["valor"] - 1 ] ); const baixado = ehBaixado( linha[ mapa["baixado"] - 1 ] ); if ( tipo === "contas a receber" || tipo === "receber" || tipo === "receita" ) { receber += valor; if (baixado) { recebido += valor; } } else if ( tipo === "contas a pagar" || tipo === "pagar" || tipo === "despesa" ) { pagar += valor; if (baixado) { pago += valor; } } } ); } /* * Saldo financeiro: * recebido - pago */ const saldo = recebido - pago; return { sucesso: true, receber: receber, pagar: pagar, recebido: recebido, pago: pago, saldo: saldo }; } /************************************************************ * DESCRIÇÕES ************************************************************/ function listarDescricoes() { const planilha = obterPlanilha(); const aba = planilha.getSheetByName( NOME_ABA_DESCRICOES ); if (!aba) { /* * Se a aba não existir, * retorna lista vazia em vez * de derrubar o sistema. */ return { sucesso: true, lista: [] }; } const ultimaLinha = aba.getLastRow(); if (ultimaLinha < 2) { return { sucesso: true, lista: [] }; } const valores = aba .getRange( 2, 1, ultimaLinha - 1, 1 ) .getValues(); const lista = []; valores.forEach( function(linha) { const descricao = String( linha[0] || "" ).trim(); if ( descricao !== "" ) { lista.push( descricao ); } } ); /* * Remove duplicadas. */ const unicas = [ ...new Set(lista) ]; return { sucesso: true, lista: unicas }; } /************************************************************ * VALIDAÇÃO ************************************************************/ function validarLancamento( competencia, tipo, descricao, valor, vencimento ) { if (!competencia) { throw new Error( "Informe a competência." ); } if ( !/^\d{4}-\d{2}$/.test( competencia ) ) { throw new Error( "Competência inválida. " + "Use o formato AAAA-MM." ); } if (!tipo) { throw new Error( "Selecione o tipo." ); } if (!descricao) { throw new Error( "Informe a descrição." ); } if ( !valor || valor <= 0 ) { throw new Error( "Informe um valor maior que zero." ); } if (!vencimento) { throw new Error( "Informe o vencimento." ); } if ( !/^\d{4}-\d{2}-\d{2}$/.test( vencimento ) ) { throw new Error( "Vencimento inválido." ); } } /************************************************************ * CONVERTER NÚMERO ************************************************************/ function converterNumero( valor ) { if ( valor === null || valor === undefined || valor === "" ) { return 0; } if ( typeof valor === "number" ) { return valor; } let texto = String(valor) .trim() .replace( /R\$/gi, "" ) .replace( /\s/g, "" ); /* * Exemplo: * 1.234,56 -> 1234.56 */ if ( texto.indexOf(",") !== -1 ) { texto = texto .replace( /\./g, "" ) .replace( ",", "." ); } const numero = Number(texto); return isNaN(numero) ? 0 : numero; } /************************************************************ * CONVERTER DATA PARA PLANILHA ************************************************************/ function converterDataPlanilha( valor ) { if (!valor) { return ""; } const texto = String(valor).trim(); /* * AAAA-MM-DD */ if ( /^\d{4}-\d{2}-\d{2}$/.test( texto ) ) { const partes = texto.split("-"); return new Date( Number(partes[0]), Number(partes[1]) - 1, Number(partes[2]) ); } /* * DD/MM/AAAA */ if ( /^\d{2}\/\d{2}\/\d{4}$/.test( texto ) ) { const partes = texto.split("/"); return new Date( Number(partes[2]), Number(partes[1]) - 1, Number(partes[0]) ); } return valor; } /************************************************************ * FORMATAR DATA ************************************************************/ function formatarData( valor ) { if (!valor) { return ""; } if ( Object.prototype.toString.call(valor) === "[object Date]" ) { if ( isNaN(valor.getTime()) ) { return ""; } return Utilities.formatDate( valor, Session.getScriptTimeZone(), "dd/MM/yyyy" ); } const texto = String(valor).trim(); /* * Já está DD/MM/AAAA */ if ( /^\d{2}\/\d{2}\/\d{4}$/.test( texto ) ) { return texto; } /* * Está AAAA-MM-DD */ if ( /^\d{4}-\d{2}-\d{2}$/.test( texto ) ) { const partes = texto.split("-"); return ( partes[2] + "/" + partes[1] + "/" + partes[0] ); } return texto; } /************************************************************ * CONVERTER COMPETÊNCIA ************************************************************/ function converterCompetencia( valor ) { if (!valor) { return ""; } if ( Object.prototype.toString.call(valor) === "[object Date]" ) { if ( isNaN(valor.getTime()) ) { return ""; } return Utilities.formatDate( valor, Session.getScriptTimeZone(), "yyyy-MM" ); } const texto = String(valor).trim(); /* * AAAA-MM */ if ( /^\d{4}-\d{2}$/.test( texto ) ) { return texto; } /* * MM/AAAA */ if ( /^\d{2}\/\d{4}$/.test( texto ) ) { const partes = texto.split("/"); return ( partes[1] + "-" + partes[0] ); } /* * MM-AAAA */ if ( /^\d{2}-\d{4}$/.test( texto ) ) { const partes = texto.split("-"); return ( partes[1] + "-" + partes[0] ); } return texto; } /************************************************************ * STATUS "BAIXADO" ************************************************************/ function ehBaixado( valor ) { const texto = String( valor || "" ) .trim() .toLowerCase(); return ( texto === "sim" || texto === "baixado" || texto === "true" || texto === "1" || texto === "s" ); } /************************************************************ * NORMALIZAR "BAIXADO" ************************************************************/ function normalizarBaixado( valor ) { return ehBaixado(valor) ? "Sim" : "Não"; }
